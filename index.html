Google Sheet 雲端資料庫連接中心

支援班別名單、題庫、學生分數及分享連結的雙向 Google Sheet 同步

方式一：連結 Google Sheet Web App (即時讀寫分數、題目及分享連結)

在你的 Google Sheet 點擊「擴充功能 > Apps Script」，貼上以下代碼並部署為「網頁應用程式 (所有人可存取)」，即可實現真實雲端同步：

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const type = data.type || "unknown";
    const payload = data.payload || {};
    let sheet = ss.getSheetByName(type);
    if (!sheet) sheet = ss.insertSheet(type);
    const now = new Date();
    sheet.appendRow([now, JSON.stringify(payload), type]);
    return ContentService.createTextOutput(JSON.stringify({status:"ok", type:type}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status:"error", message:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const type = (e && e.parameter && e.parameter.type) ? e.parameter.type : "grades";
  const sheet = ss.getSheetByName(type);
  if (!sheet) return ContentService.createTextOutput(JSON.stringify({status:"no_sheet"}))
    .setMimeType(ContentService.MimeType.JSON);
  const data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({status:"ok", type:type, data:data}))
    .setMimeType(ContentService.MimeType.JSON);
}
